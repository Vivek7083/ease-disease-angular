const ACTIVITY = ['Mostly sitting', 'Lightly active', 'Moderately active', 'Very active'] as const;
const STRESS = ['Low', 'Moderate', 'High', 'Very high'] as const;

const notBlank = (c: AbstractControl): ValidationErrors | null => (String(c.value ?? '').trim() ? null : { required: true });
const email = (c: AbstractControl): ValidationErrors | null =>
  /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(String(c.value ?? '').trim()) ? null : { email: true };
const phone = (c: AbstractControl): ValidationErrors | null =>
  String(c.value ?? '').replace(/\D/g, '').length >= 10 ? null : { phone: true };
const age = (c: AbstractControl): ValidationErrors | null => {
  const n = Number(c.value);
  return Number.isInteger(n) && n >= 1 && n <= 120 ? null : { age: true };
};
const atLeastOne = (c: AbstractControl): ValidationErrors | null => ((c.value as string[]).length > 0 ? null : { required: true });

/** Which controls each of the three steps checks before moving on. */
const STEP_FIELDS = [
  ['name', 'phone', 'email', 'age', 'city'],
  ['concerns', 'symptoms', 'duration'],
  ['diet', 'sleep', 'activity', 'stress', 'consent'],
] as const;

const makeForm = () =>
  new FormGroup({
    name: new FormControl('', { nonNullable: true, validators: [notBlank] }),
    phone: new FormControl('', { nonNullable: true, validators: [phone] }),
    email: new FormControl('', { nonNullable: true, validators: [email] }),
    age: new FormControl('', { nonNullable: true, validators: [age] }),
    city: new FormControl('', { nonNullable: true, validators: [notBlank] }),
    gender: new FormControl('', { nonNullable: true }),
    concerns: new FormControl<string[]>([], { nonNullable: true, validators: [atLeastOne] }),
    symptoms: new FormControl('', { nonNullable: true, validators: [notBlank] }),
    duration: new FormControl('', { nonNullable: true, validators: [Validators.required] }),
    diagnosed: new FormControl('', { nonNullable: true }),
    meds: new FormControl('', { nonNullable: true }),
    diet: new FormControl('', { nonNullable: true, validators: [Validators.required] }),
    sleep: new FormControl('', { nonNullable: true, validators: [Validators.required] }),
    activity: new FormControl('', { nonNullable: true, validators: [Validators.required] }),
    stress: new FormControl('', { nonNullable: true, validators: [Validators.required] }),
    consent: new FormControl(false, { nonNullable: true, validators: [Validators.requiredTrue] }),
  });

  protected readonly stressOptions = STRESS;

  protected readonly step = signal(0);
  protected readonly attempted = signal([false, false, false]);
  protected readonly selectedConcerns = signal<readonly string[]>([]);
  protected readonly firstName = signal('');
  protected readonly demoNote = signal('');

  protected readonly form = makeForm();


  constructor() {
    effect(() => {
      const wantOpen = this.modal.isOpen();
      untracked(() => {
        const dialog = this.dlg()?.nativeElement;
        if (!dialog) {
          return;
        }
        if (wantOpen && !dialog.open) {
          this.reset();
          dialog.showModal();
          document.body.style.overflow = 'hidden';
          this.goTo(0); // start in the first field, not on the close button
        } else if (!wantOpen && dialog.open) {
          dialog.close();
        }
      });
    });
  }

  /** Fires for every way the dialog closes: the X, Escape, a click outside, or close(). */
  protected onNativeClose(): void {
    document.body.style.removeProperty('overflow');
    this.modal.close();
  }

  /** A click on the dialog itself (not the card inside it) is a click outside the form. */
  protected onDialogClick(event: MouseEvent): void {
    if (event.target === this.dlg()?.nativeElement) {
      this.modal.close();
    }
  }

  /** True when this field should show its error: it is invalid and was touched, or its step was already attempted. */
  protected invalid(name: FieldName): boolean {
    const control = this.form.controls[name];
    const stepIndex = STEP_FIELDS.findIndex((fields) => (fields as readonly string[]).includes(name));
    return control.invalid && (control.touched || (stepIndex >= 0 && this.attempted()[stepIndex]));
  }

  protected toggleConcern(concern: string): void {
    const current = this.selectedConcerns();
    const next = current.includes(concern) ? current.filter((c) => c !== concern) : [...current, concern];
    this.selectedConcerns.set(next);
    this.form.controls.concerns.setValue([...next]);
    this.form.controls.concerns.markAsTouched();
  }

  protected back(): void {
    this.goTo(this.step() - 1);
  }

  protected next(): void {
    const current = this.step();
    if (current === 3) {
      this.continueToPayment();
      return;
    }

    const fields = STEP_FIELDS[current];
    this.attempted.update((a) => a.map((v, i) => v || i === current));
    let ok = true;
    for (const name of fields) {
      const control = this.form.controls[name];
      control.markAsTouched();
      ok = ok && control.valid;
    }
    if (!ok) {
      afterNextRender(() => this.dlg()?.nativeElement.querySelector<HTMLElement>('[aria-invalid="true"], fieldset[data-invalid="true"] input')?.focus(), { injector: this.injector });
      return;
    }

    if (current < 2) {
      this.goTo(current + 1);
    } else {
      this.submit();
    }
  }

  /**
   * DEMO MODE: nothing is sent. When email delivery is wired up (Web3Forms),
   * send this.form.getRawValue() here and only go to the thank-you screen on
   * success, showing an error and keeping the answers if it fails.
   */
  private submit(): void {
    this.firstName.set(this.form.controls.name.value.trim().split(/\s+/)[0] ?? '');
    this.demoNote.set('Demo mode: nothing was sent. On the live site these answers are emailed to you.');
    this.goTo(3);
  }

  /** DEMO MODE: on the live site this opens the Razorpay payment page for the 2,499 rupee consultation. */
  private continueToPayment(): void {
    this.demoNote.set('Demo mode: on the live site this button opens the ₹2,499 payment page.');
  }

  private goTo(step: number): void {
    this.step.set(step);
    afterNextRender(
      () => {
        this.body()?.nativeElement.scrollTo({ top: 0 });
        const dialog = this.dlg()?.nativeElement;
        const target = step === 3 ? dialog?.querySelector<HTMLElement>('.bk-done') : dialog?.querySelector<HTMLElement>('.bk-body input, .bk-body select, .bk-body textarea');
        target?.focus({ preventScroll: true });
      },
      { injector: this.injector },
    );
  }

  private reset(): void {
    this.form.reset();
    this.selectedConcerns.set([]);
    this.attempted.set([false, false, false]);
    this.firstName.set('');
    this.demoNote.set('');
    this.step.set(0);
  }
}
